-- and 20261101190000_records_pipeline.sql (only the columns that matter).
-- Counts follow docs/data/identity.md (51,900 groups: 10 x 5,000 + 190 x 10)
-- and docs/data/records-list-bench.md (800k records, one 400k-row tenant).
DROP TABLE IF EXISTS group_members, records, groups, principals, tenants CASCADE;
CREATE TABLE tenants (id uuid PRIMARY KEY);
CREATE TABLE groups (
    tenant_id uuid NOT NULL REFERENCES tenants(id),
    id uuid NOT NULL,
    keycloak_group_id text NOT NULL,
    path text NOT NULL,
    PRIMARY KEY (tenant_id, id),
    UNIQUE (tenant_id, keycloak_group_id));
CREATE TABLE principals (tenant_id uuid NOT NULL, id uuid NOT NULL, PRIMARY KEY (tenant_id, id));
CREATE TABLE group_members (
    tenant_id uuid NOT NULL, group_id uuid NOT NULL, principal_id uuid NOT NULL,
    PRIMARY KEY (tenant_id, group_id, principal_id),
    FOREIGN KEY (tenant_id, group_id) REFERENCES groups (tenant_id, id) ON DELETE CASCADE);
CREATE INDEX group_members_principal ON group_members (tenant_id, principal_id);
CREATE TABLE records (
    tenant_id uuid NOT NULL, id uuid NOT NULL,
    state text NOT NULL, assignment_group_id uuid,
    updated_at timestamptz NOT NULL, data jsonb NOT NULL DEFAULT '{}',
    PRIMARY KEY (tenant_id, id),
    CONSTRAINT records_assignment_group_fkey FOREIGN KEY (tenant_id, assignment_group_id)
        REFERENCES groups (tenant_id, id) ON DELETE SET NULL (assignment_group_id));
CREATE INDEX records_tenant_group_open ON records (tenant_id, assignment_group_id, updated_at DESC, id DESC)
    WHERE state <> 'closed';

-- C. R29 catalogue query, before the change (expect one row).

SELECT c.conrelid::regclass AS tbl, c.conname, c.confdeltype, c.confupdtype
FROM pg_constraint c
WHERE c.contype = 'f'
  AND c.conrelid = ANY (:'audited'::regclass[])
  AND (c.confdeltype NOT IN ('a', 'r') OR c.confupdtype NOT IN ('a', 'r'));

-- D. Expand step: drop + re-add NOT VALID in ONE transaction. Locks held?
BEGIN;
SET LOCAL lock_timeout = '5s';
ALTER TABLE records DROP CONSTRAINT records_assignment_group_fkey,
  ADD CONSTRAINT records_assignment_group_fkey FOREIGN KEY (tenant_id, assignment_group_id)
    REFERENCES groups (tenant_id, id) ON DELETE RESTRICT NOT VALID;
SELECT l.relation::regclass, l.mode FROM pg_locks l
WHERE l.locktype = 'relation' AND l.pid = pg_backend_pid()
  AND l.relation::regclass::text IN ('records', 'groups') ORDER BY 1, 2;
COMMIT;

-- Reader 1: resolver (pgstore.go groupPathsSQL), worst-case principal, 200 groups.
PREPARE r_old(uuid, uuid) AS SELECT g.id, g.path FROM group_members gm
 JOIN groups g ON g.tenant_id = gm.tenant_id AND g.id = gm.group_id
 WHERE gm.tenant_id = $1 AND gm.principal_id = $2 ORDER BY g.id;
PREPARE r_new(uuid, uuid) AS SELECT g.id, g.path FROM group_members gm
 JOIN groups g ON g.tenant_id = gm.tenant_id AND g.id = gm.group_id
 WHERE gm.tenant_id = $1 AND gm.principal_id = $2 AND g.deleted_at IS NULL ORDER BY g.id;
