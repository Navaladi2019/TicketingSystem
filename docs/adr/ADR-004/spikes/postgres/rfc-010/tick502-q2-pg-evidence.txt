===== schema.sql
\set ON_ERROR_STOP 1
DROP TABLE IF EXISTS audit_log, aud_shipped, aud_custom CASCADE;
CREATE TABLE audit_log (
  tenant_id uuid NOT NULL, id uuid NOT NULL DEFAULT gen_random_uuid(),
  occurred_at timestamptz NOT NULL DEFAULT now(), txid xid8 NOT NULL DEFAULT pg_current_xact_id(),
  subject_type text NOT NULL, subject_id uuid NOT NULL, version_after bigint, sealed_at timestamptz,
  filler text,
  CONSTRAINT audit_log_pkey PRIMARY KEY (tenant_id, occurred_at, id),
  CONSTRAINT audit_log_commit_key UNIQUE NULLS NOT DISTINCT (tenant_id, occurred_at, txid, subject_type, subject_id, version_after)
) PARTITION BY RANGE (occurred_at);
CREATE INDEX audit_log_tenant_time ON audit_log (tenant_id, occurred_at DESC, id DESC);
CREATE INDEX audit_log_subject ON audit_log (tenant_id, subject_type, subject_id, occurred_at DESC);
CREATE INDEX audit_log_unsealed ON audit_log (tenant_id, occurred_at, id) WHERE sealed_at IS NULL;
-- three young partitions, autovacuum off so only we change their stats
CREATE TABLE p_never     PARTITION OF audit_log FOR VALUES FROM ('2026-01-01') TO ('2026-02-01') WITH (autovacuum_enabled=false);
CREATE TABLE p_an_empty  PARTITION OF audit_log FOR VALUES FROM ('2026-02-01') TO ('2026-03-01') WITH (autovacuum_enabled=false);
CREATE TABLE p_vac_empty PARTITION OF audit_log FOR VALUES FROM ('2026-03-01') TO ('2026-04-01') WITH (autovacuum_enabled=false);
ALTER TABLE p_never ALTER COLUMN occurred_at SET (n_distinct=1);
ALTER TABLE p_an_empty ALTER COLUMN occurred_at SET (n_distinct=1);
ALTER TABLE p_vac_empty ALTER COLUMN occurred_at SET (n_distinct=1);

-- the TICK-428 function body verbatim (7b5e388), as two copies: as shipped, and with RFC-010 route 1c
CREATE OR REPLACE FUNCTION cc_shipped() RETURNS trigger LANGUAGE plpgsql SECURITY INVOKER SET search_path = pg_catalog, public AS $f$
DECLARE v_tenant uuid; v_id uuid; v_type text := TG_ARGV[0]; v_txid xid8 := pg_current_xact_id(); v_at timestamptz := now();
BEGIN
  IF TG_OP = 'DELETE' THEN v_tenant := OLD.tenant_id; v_id := OLD.id; ELSE v_tenant := NEW.tenant_id; v_id := NEW.id; END IF;
  PERFORM 1 FROM audit_log WHERE tenant_id = v_tenant AND occurred_at = v_at AND txid = v_txid AND subject_type = v_type AND subject_id = v_id LIMIT 1;
  IF NOT FOUND THEN RAISE EXCEPTION 'audit_missing % %', v_type, v_id USING ERRCODE='23514'; END IF;
  RETURN NULL;
END $f$;
CREATE OR REPLACE FUNCTION cc_custom() RETURNS trigger LANGUAGE plpgsql SECURITY INVOKER
  SET search_path = pg_catalog, public SET plan_cache_mode = force_custom_plan AS $f$
DECLARE v_tenant uuid; v_id uuid; v_type text := TG_ARGV[0]; v_txid xid8 := pg_current_xact_id(); v_at timestamptz := now();
BEGIN
  IF TG_OP = 'DELETE' THEN v_tenant := OLD.tenant_id; v_id := OLD.id; ELSE v_tenant := NEW.tenant_id; v_id := NEW.id; END IF;
  PERFORM 1 FROM audit_log WHERE tenant_id = v_tenant AND occurred_at = v_at AND txid = v_txid AND subject_type = v_type AND subject_id = v_id LIMIT 1;
  IF NOT FOUND THEN RAISE EXCEPTION 'audit_missing % %', v_type, v_id USING ERRCODE='23514'; END IF;
  RETURN NULL;
END $f$;
CREATE TABLE aud_shipped (tenant_id uuid NOT NULL, id uuid NOT NULL, PRIMARY KEY (tenant_id, id));
CREATE TABLE aud_custom  (tenant_id uuid NOT NULL, id uuid NOT NULL, PRIMARY KEY (tenant_id, id));
CREATE CONSTRAINT TRIGGER cc AFTER INSERT ON aud_shipped DEFERRABLE INITIALLY DEFERRED FOR EACH ROW EXECUTE FUNCTION cc_shipped('record');
CREATE CONSTRAINT TRIGGER cc AFTER INSERT ON aud_custom  DEFERRABLE INITIALLY DEFERRED FOR EACH ROW EXECUTE FUNCTION cc_custom('record');
SELECT proname, proconfig FROM pg_proc WHERE proname LIKE 'cc_%' ORDER BY 1;
