+	p := e.tokens[tok]
+	def := e.region.defs[c.key]
+	var b strings.Builder
+	explain := func(tx pgx.Tx, label, sql string, args ...any) {
+		rows, err := tx.Query(ctx, "EXPLAIN (ANALYZE, BUFFERS) "+sql, args...)
+		if err != nil {
+			fmt.Fprintf(&b, "%s: EXPLAIN error %v\n", label, err)
+			return
+		}
+		defer rows.Close()
+		fmt.Fprintf(&b, "\n%s:\n```\n", label)
+		for rows.Next() {
+			var l string
+			if err := rows.Scan(&l); err != nil {
+				t.Fatal(err)
+			}
+			b.WriteString(l + "\n")
+		}
+		b.WriteString("```\n")
+	}
+	err := tenancy.InTenantTx(ctx, e.pool, p.TenantID, func(tx pgx.Tx) error {
+		pol, err := authz.PolicyStore{}.Current(ctx, tx, c.key)
+		if err != nil {
+			return err
+		}
+		bound, err := authz.ResolveUnder(ctx, tx, pol, p)
+		if err != nil {
+			return err
+		}
+		var ne int64
+		if err := tx.QueryRow(ctx, "SELECT live_records FROM entity_record_estimates WHERE tenant_id = $1 AND entity_definition_id = $2", p.TenantID, def).Scan(&ne); err != nil {
+			return err
+		}
+		k := authz.KStar(perfRegionPage, ne, true)
+		cap := int64(math.Ceil(float64(ne) / spikeAlpha))
+		units := make([]uuid.UUID, 0, len(c.pres))
+		for _, pr := range c.pres {
+			units = append(units, e.region.units[pr])
+		}
+		fmt.Fprintf(&b, "N̂_e %d, K* %d, alpha %.1f, cap ceil(N̂_e/alpha) = %d\n", ne, k, spikeAlpha, cap)
+		explain(tx, "folded probe (|S| capped at N̂_e/alpha, P3 in CASE ELSE)", authz.FoldedProbeStatement(k), p.TenantID, authz.UUIDTexts(sortedUUIDs(units)), def, cap)
+		spikeMode("R")
+		shapes, err := authz.ChooseListShapes(ctx, tx, bound, p, def, perfRegionPage)
+		if err != nil {
+			return err
+		}
+		read := authz.ReadListShaped(bound, p, "r", 2, shapes)
+		q := pipeline.ListQuery{TenantID: p.TenantID, DefinitionID: def, Where: "true", Args: read.Args, Branches: read.Branches,
+			CountBranches: read.CountBranches, Materialize: read.Materialize, SortColumn: "updated_at", Desc: true, Limit: perfRegionPage}
+		if read.Branches == nil {
+			q.Where, q.CountWhere = read.Where, read.CountWhere
+		}
+		for _, d := range read.Drivers {
+			q.CountDrivers = append(q.CountDrivers, pipeline.CountDriver{From: d.From, Key: d.Key})
+		}
+		args := append([]any{q.TenantID, q.DefinitionID}, q.Args...)
+		explain(tx, "count", store.CountStatement(q), args...)
+		sql, pargs, err := store.PageStatement(q)
+		if err != nil {
+			return err
+		}
+		explain(tx, "page", sql, pargs...)
+		return nil
+	})
+	if err != nil {
+		fmt.Fprintf(&b, "error: %v\n", err)
+	}
+	return b.String()
+}

+func sortedUUIDs(u []uuid.UUID) []uuid.UUID {
+	out := append([]uuid.UUID(nil), u...)
+	for i := range out {
+		for j := i + 1; j < len(out); j++ {
+			if strings.Compare(out[j].String(), out[i].String()) < 0 {
+				out[i], out[j] = out[j], out[i]
+			}
+		}
+	}
+	return out
+}
